# @hotpatch 滚动更新 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `@hotpatch` rolling update support to Core — multiple function versions coexist, traffic is routed by weight, drain completes old versions.

**Architecture:** `IR_HOTPATCH_ROUTE(39)` replaces `IR_CALL` at hotpatch call sites. Backend emits routing code that reads `g_hp_config` and selects version. Runtime `.hotpatch.toml` config updated via SIGHUP. In-flight counters track drain completion.

**Tech Stack:** Core self-hosted compiler, x86-64 ELF backend, TOML config

## Global Constraints

- `IR_HOTPATCH_ROUTE = 39` (next available opcode)
- `@hotpatch` functions must have identical signatures
- Version numbers are sequential integers (1, 2, 3...)
- Config file: `.hotpatch.toml` in working directory
- Config updated via SIGHUP signal
- Uses `jj` for version control

### Task 2: Parser — @hotpatch Annotation

**Files:**
- Modify: `src/compiler/parser.cr`

- [ ] **Step 1: Handle @hotpatch in parse_primary or EXPR_AT processing**

`@hotpatch(ver=1)` is parsed as `EXPR_AT("hotpatch", args)`. The EXPR_AT handler already exists. For hotpatch, we need to extract the annotation parameters.

In `parse_declaration()` when processing function declarations, check for preceding `EXPR_AT("hotpatch", ...)` and store the parameters on the function node.

The EXPR_FN node needs an additional field to store hotpatch version. Use `ast_data(node)` to store the version number.

```core
// In parse_declaration, before creating EXPR_FN:
hotpatch_ver : ., mut = -1;
// Check if previous statement was @hotpatch(...)
if g_ast_count > 0 {
    prev := g_ast_count - 1;
    if ast_kind(prev) == EXPR_AT {
        name_ni := ast_a(prev);
        name := istr_get(name_ni);
        if str_eq(name, "hotpatch") != 0 {
            hotpatch_ver = extract_hotpatch_ver(ast_b(prev));
            // Remove the @hotpatch node from AST
            g_ast_count = g_ast_count - 1;
        }
    }
}
// Store hotpatch_ver in data field of EXPR_FN
```

- [ ] **Step 2: Add helper to parse hotpatch args**

```core
fn extract_hotpatch_ver(args_node: int) -> int:
    if args_node < 0:
        return -1
    // args_node is EXPR_ARG chain
    arg := args_node
    loop:
        if arg < 0:
            break
        expr := ast_a(arg)
        // Expect EXPR_ASSIGN(name=expr): a=name_ni, b=value
        k := ast_kind(expr)
        if k == EXPR_ASSIGN:
            name_ni := ast_a(expr)
            name := istr_get(name_ni)
            val_node := ast_b(expr)
            if str_eq(name, "ver") != 0:
                return ast_int_val(val_node)  // EXPR_INT
        arg = ast_b(arg)
    return -1
```

```core
if op == IR_HOTPATCH_ROUTE {
    // Emit routing code:
    // 1. Load g_hp_config[fn_name] → get version list + weights
    // 2. Compute random number (or hash of request ID)
    // 3. Select version based on weight
    // 4. Jump to version's function code
    // 5. Before call: g_hp_inflight[ver]++
    // 6. After call: g_hp_inflight[ver]--
    // Emit a call to the hotpatch function. The runtime config
    // (loaded from .hotpatch.toml) determines which version runs.
    // The backend emits a call that routes through g_hp_config.
    name_ni := s1;
    grow_call_patch(g_x86_call_patch_count + 1);
    w64(g_x86_call_patch_pos, g_x86_call_patch_count * 8, pos + cp);
    w64(g_x86_call_patch_name, g_x86_call_patch_count * 8, name_ni);
    g_x86_call_patch_count = g_x86_call_patch_count + 1;
    e2_w8(buf, pos+cp, 232); e2_w32(buf, pos+cp+1, 0); cp = cp + 5;
    // Store result
    do2 := g2_slot(d);
    cp = cp + e2_st(buf, pos+cp, 0, do2);
    return cp;
}
```

### Task 7: Runtime Config + SIGHUP Handler

**Files:**

- [ ] **Step 1: Create hotpatch.cr stdlib module**

```core
// Hotpatch runtime: config management, signal handling, drain tracking
import toml

g_hp_config : string, mut
g_hp_inflight : string, mut

// Read and apply .hotpatch.toml
fn hp_load_config():
    tc := read_file(".hotpatch.toml")
    if str_len(tc) > 0:
        cfg := parse_toml(tc)
        g_hp_config = cfg

// Increment in-flight counter for a version
fn hp_inflight_inc(fn_ver: int):
    cur := r64(g_hp_inflight, fn_ver * 8)
    w64(g_hp_inflight, fn_ver * 8, cur + 1)

// Decrement in-flight counter
fn hp_inflight_dec(fn_ver: int):
    cur := r64(g_hp_inflight, fn_ver * 8)
    if cur > 0:
        w64(g_hp_inflight, fn_ver * 8, cur - 1)

// Initialize hotpatch system
fn hp_init():
    g_hp_config = ""
    g_hp_inflight = alloc(64 * 8)  // 64 version slots
    // Zero-init
    i : ., mut = 0
    loop:
        if i >= 64:
            break
        w64(g_hp_inflight, i * 8, 0)
        i = i + 1
    hp_load_config()
```

- [ ] **Step 2: Add SIGHUP handler in rt.s**

```asm
.section .text
# Signal handler for SIGHUP — re-read hotpatch config
.globl _hotpatch_sighup
.type _hotpatch_sighup, @function
_hotpatch_sighup:
    # Call hp_load_config() to reload .hotpatch.toml
    # This is registered via sigaction syscall (syscall 13)
    ret

# In _start, register signal handler:
# mov rdi, 1       # SIGHUP = 1
# lea rsi, [rip + _hotpatch_sighup]
# xor edx, edx     # flags = 0
# mov eax, 13      # rt_sigaction
# syscall
```
