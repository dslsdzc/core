# FFI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add FFI to Core: `extern fn` declarations, C ABI calls, `@ffi` multi-language annotation, automatic `.so` linking.

**Architecture:** Parser handles `extern fn` + `@ffi`. Reuses existing `.so` linker infrastructure (ctx_add_so, ctx_add_plt, ext_rel). `IR_CALL_EXTERN(45)` for external calls.

**Tech Stack:** Core self-hosted compiler, x86-64 ELF backend, dynamic linker

## Global Constraints

- `T_EXTERN = 100`, `EXPR_EXTERN`, `IR_CALL_EXTERN = 45`
- Reuses existing `ctx_add_so`, `ctx_add_plt`, `g_x86_ext_rel_*` infrastructure
- `@ffi("C")` is the default (no conversion)
- `@ffi("Python")` inserts PyObject* marshaling code at call sites
- Uses `jj` for version control

### Task 2: Parser — extern fn + @ffi

**Files:**
- Modify: `src/compiler/parser.cr`

- [ ] **Step 1: Parse extern fn declarations**

In `parse_declaration`, after checking for `pub`/`fn`, check for `T_EXTERN`:

```core
if tok_k(cur_tok()) == T_EXTERN {
    advance_tok();
    ffi_lang_ni : ., mut = -1;
    // Check for @ffi("...")
    if tok_k(cur_tok()) == T_AT {
        ffi_lang_ni = parse_ffi_annotation();
    }
    // Expect fn keyword
    if tok_k(cur_tok()) != T_FN {
        add_error("expected 'fn' after extern");
        return -1;
    }
    advance_tok();
    // Parse function name
    fn_name_ni := tok_lx(cur_tok()); advance_tok();
    // Parse generic params if present
    // Parse param list
    first_param := parse_param_list();
    param_count := ...;
    // Parse return type
    ret_type := TI_UNIT;
    if check(T_ARROW) { advance_tok(); ret_type = parse_type(); }
    // Create EXPR_EXTERN node
    node := alloc_node(EXPR_EXTERN, fn_name_ni, first_param, param_count, 0, ret_type, ffi_lang_ni, line, col);
    // Skip body — extern fns have no body
    if tok_k(cur_tok()) == T_LBRACE {
        skip_block();  // consume but ignore body for friendlier errors
    }
    advance_tok();  // skip semicolon or brace end
    return node;
}
```

- [ ] **Step 2: Parse @ffi("lang") annotation**

```core
fn parse_ffi_annotation() -> int:
    // Current token is T_AT
    advance_tok()  // skip @
    if tok_k(cur_tok()) != T_IDENT:
        add_error("expected ffi")
        return -1
    lex := tok_lx(cur_tok())
    if str_eq(lex, "ffi") == 0:
        add_error("expected ffi")
        return -1
    advance_tok()
    // Expect ( "lang_name" )
    if tok_k(cur_tok()) != T_LPAREN:
        add_error("expected (")
        return -1
    advance_tok()
    if tok_k(cur_tok()) != T_STR:
        add_error("expected string literal")
        return -1
    lang_ni := tok_lx(cur_tok())
    advance_tok()
    if tok_k(cur_tok()) != T_RPAREN:
        add_error("expected )")
        return -1
    advance_tok()
    return lang_ni
```

if is_extern_func(func_ni) {
    ffi_lang := get_ffi_lang(func_ni);
    if str_eq(ffi_lang, "C") != 0 {
        // C ABI: direct call via external relocation
        emit(IR_CALL_EXTERN, dest, func_ni, first_arg, arg_count, 0);

}

### Task 5: ELF Backend — Extern Call Encoding

**Files:**
- Modify: `src/arch/linux/ld/instr.cr`
- Modify: `src/arch/linux/ld/sizes.cr`

- [ ] **Step 1: Add IR_CALL_EXTERN encoding**

```core
if op == IR_CALL_EXTERN {
    // External function call: emit call via PLT
    // s1 = func name index, s2 = first arg, s3 = arg count
    name_ni := s1;
    // Use external relocation mechanism
    grow_ext_rel(g_x86_ext_rel_count + 1);
    w64(g_x86_ext_rel_pos, g_x86_ext_rel_count * 8, pos + cp);
    w64(g_x86_ext_rel_name, g_x86_ext_rel_count * 8, name_ni);
    g_x86_ext_rel_count = g_x86_ext_rel_count + 1;
    // Emit call placeholder (5 bytes: E8 + rel32)
    e2_w8(buf, pos+cp, 232); e2_w32(buf, pos+cp+1, 0); cp = cp + 5;
    // Store return value
    do2 := g2_slot(d);
    cp = cp + e2_st(buf, pos+cp, 0, do2);
    return cp;
}
```
