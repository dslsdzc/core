# FFI 实现计划

> 受众：执行本计划的 agentic worker。必需子技能：`superpowers:subagent-driven-development`（推荐）或 `superpowers:executing-plans`，按任务逐项实施。进度以复选框（`- [ ]`）跟踪。

目标：为 Core 增加 FFI，包括 `extern fn` 声明、C ABI 调用、`@ffi` 多语言注解和自动链接 `.so`。

架构：Parser 处理 `extern fn` 与 `@ffi`。复用既有的 `.so` 链接基础设施（ctx_add_so、ctx_add_plt、ext_rel）。外部调用使用 `IR_CALL_EXTERN(45)`。

技术栈：Core 自举编译器、x86-64 ELF 后端、动态链接器

## 全局约束

- `T_EXTERN = 100`、`EXPR_EXTERN`、`IR_CALL_EXTERN = 45`
- 复用既有的 `ctx_add_so`、`ctx_add_plt`、`g_x86_ext_rel_*` 基础设施
- `@ffi("C")` 是默认值，不做转换
- `@ffi("Python")` 在调用点插入 PyObject* 编组代码
- 版本控制使用 `jj`

### 任务 2：Parser 处理 extern fn 与 @ffi

涉及文件：
- 修改：`src/compiler/parser.cr`

- [ ] 步骤 1：解析 extern fn 声明

在 `parse_declaration` 中，确认 `pub`/`fn` 之后，再检查 `T_EXTERN`：

```core
if tok_k(cur_tok()) == T_EXTERN {
    advance_tok();
    ffi_lang_ni : ., mut = -1;
    // Check for @ffi("...")
    if tok_k(cur_tok()) == T_AT {
        ffi_lang_ni = parse_ffi_annotation();
    }
    // Expect fn keyword
    if tok_k(cur_tok()) != T_FN {
        add_error("expected 'fn' after extern");
        return -1;
    }
    advance_tok();
    // Parse function name
    fn_name_ni := tok_lx(cur_tok()); advance_tok();
    // Parse generic params if present
    // Parse param list
    first_param := parse_param_list();
    param_count := ...;
    // Parse return type
    ret_type := TI_UNIT;
    if check(T_ARROW) { advance_tok(); ret_type = parse_type(); }
    // Create EXPR_EXTERN node
    node := alloc_node(EXPR_EXTERN, fn_name_ni, first_param, param_count, 0, ret_type, ffi_lang_ni, line, col);
    // Skip body — extern fns have no body
    if tok_k(cur_tok()) == T_LBRACE {
        skip_block();  // consume but ignore body for friendlier errors
    }
    advance_tok();  // skip semicolon or brace end
    return node;
}
```

- [ ] 步骤 2：解析 @ffi("lang") 注解

```core
fn parse_ffi_annotation() -> int:
    // Current token is T_AT
    advance_tok()  // skip @
    if tok_k(cur_tok()) != T_IDENT:
        add_error("expected ffi")
        return -1
    lex := tok_lx(cur_tok())
    if str_eq(lex, "ffi") == 0:
        add_error("expected ffi")
        return -1
    advance_tok()
    // Expect ( "lang_name" )
    if tok_k(cur_tok()) != T_LPAREN:
        add_error("expected (")
        return -1
    advance_tok()
    if tok_k(cur_tok()) != T_STR:
        add_error("expected string literal")
        return -1
    lang_ni := tok_lx(cur_tok())
    advance_tok()
    if tok_k(cur_tok()) != T_RPAREN:
        add_error("expected )")
        return -1
    advance_tok()
    return lang_ni
```

if is_extern_func(func_ni) {
    ffi_lang := get_ffi_lang(func_ni);
    if str_eq(ffi_lang, "C") != 0 {
        // C ABI: direct call via external relocation
        emit(IR_CALL_EXTERN, dest, func_ni, first_arg, arg_count, 0);

### 任务 5：ELF 后端的 extern 调用编码

涉及文件：
- 修改：`src/arch/linux/ld/instr.cr`
- 修改：`src/arch/linux/ld/sizes.cr`

- [ ] 步骤 1：添加 IR_CALL_EXTERN 编码

```core
if op == IR_CALL_EXTERN {
    // External function call: emit call via PLT
    // s1 = func name index, s2 = first arg, s3 = arg count
    name_ni := s1;
    // Use external relocation mechanism
    grow_ext_rel(g_x86_ext_rel_count + 1);
    w64(g_x86_ext_rel_pos, g_x86_ext_rel_count * 8, pos + cp);
    w64(g_x86_ext_rel_name, g_x86_ext_rel_count * 8, name_ni);
    g_x86_ext_rel_count = g_x86_ext_rel_count + 1;
    // Emit call placeholder (5 bytes: E8 + rel32)
    e2_w8(buf, pos+cp, 232); e2_w32(buf, pos+cp+1, 0); cp = cp + 5;
    // Store return value
    do2 := g2_slot(d);
    cp = cp + e2_st(buf, pos+cp, 0, do2);
    return cp;
}
```
